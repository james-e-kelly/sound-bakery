116 [[nodiscard]] auto write(void* message, std::size_t messageSize) noexcept -> sbk_status
117 {
118 if (message == nullptr || messageSize == 0)
119 {
120 return SBK_ERR_INVALID_PARAMETER;
121 }
122
123 while (true)
124 {
125 const std::size_t read = m_readIndex.load(std::memory_order_relaxed);
126 std::size_t reserveWrite = m_reserveWriteIndex.load(std::memory_order_relaxed);
127
128 const bool isFull = reserveWrite - read == m_capacity;
129 const bool hasEnoughSpace = reserveWrite + messageSize - read <= m_capacity;
130
131 if (isFull || !hasEnoughSpace)
132 {
133 return SBK_ERR_BAKERY;
134 }
135
136 if (m_reserveWriteIndex.compare_exchange_weak(reserveWrite, reserveWrite + messageSize, std::memory_order_relaxed))
137 {
138 std::memcpy(m_buffer + (reserveWrite & m_mask), message, messageSize);
139
140 std::size_t expected = reserveWrite;
141 while (!m_committedWriteIndex.compare_exchange_weak(expected, reserveWrite + messageSize, std::memory_order_release, std::memory_order_relaxed))
142 {
143 expected = reserveWrite;
144 std::this_thread::yield();
145 }
146 return SBK_SUCCESS;
147 }
148 }

156 [[nodiscard]] auto read(void* outBuffer, std::size_t readBytes) noexcept -> sbk_status
157 {
158 if (readBytes == 0 || outBuffer == nullptr)
159 {
160 return SBK_ERR_INVALID_PARAMETER;
161 }
162
163 const std::size_t read = m_readIndex.load(std::memory_order_relaxed);
164 const std::size_t committedWrite = m_committedWriteIndex.load(std::memory_order_acquire);
165
166 const bool bufferEmpty = read == committedWrite;
167 const bool canReadBytes = read + readBytes <= committedWrite;
168
169 if (bufferEmpty || !canReadBytes)
170 {
171 return SBK_ERR_BAKERY;
172 }
173
174 std::memcpy(outBuffer, m_buffer + (read & m_mask), readBytes);
175
176 m_readIndex.store(read + readBytes, std::memory_order_relaxed);
177
178 return SBK_SUCCESS;
179 }
