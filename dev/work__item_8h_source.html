56 [[nodiscard]] static auto resume_owning(std::coroutine_handle<> handle, const char* fiberName) -> work_item
57 {
58 // Ownership handoff: while the lambda lives, the captured unique_coroutine owns the frame,
59 // so dropping the item unrun destroys it. Running the lambda transfers the raw handle to
60 // resume(), which either completes the frame (self-destroys) or hands it to whatever
61 // executor the resume suspends onto next.
62 return work_item{[coro = unique_coroutine<>{handle}, fiberName]() mutable
63 { resume(coro.release(), fiberName); }};
64 }
