86 template <pod U>
87 [[nodiscard]] auto write_message(T type, const U& message) noexcept -> sbk_status
88 {
89 SBK_STATUS_CHECK_MSG(to_numeric(type) != s_skipFlag, SBK_ERR_INVALID_PARAMETER, "Users should not create messages with a flag equal to s_skipFlag. This should only be used by the message_queue to automically wrap around the end of the buffer");
90
91 std::uint8_t* messageBuffer{};
92 std::uint8_t* paddingBuffer{};
93 std::size_t paddingSize{};
94 std::size_t reserveIndex{};
95
96 message_header header{.m_type = to_numeric(type), .m_payloadSize = sizeof(U)};
97 SBK_STATUS_TRY_C(m_ringBuffer.reserve_write(&messageBuffer, sizeof(message_header) + header.m_payloadSize, &reserveIndex, &paddingBuffer, &paddingSize));
98
99 // If we were given padding, it means the buffer is split
100 // Add a skip message into end of the buffer. Write the actual message as normal
101 if (paddingSize > 0U)
102 {
103 BOOST_ASSERT(paddingSize >= sizeof(message_header));
104 BOOST_ASSERT(paddingSize <= std::numeric_limits<message_size_type>::max());
105 BOOST_ASSERT(paddingBuffer > messageBuffer);
106 message_header skipHeader{.m_type = s_skipFlag, .m_payloadSize = static_cast<message_size_type>(paddingSize) - sizeof(message_header)};
107 std::memcpy(paddingBuffer, &skipHeader, sizeof(message_header));
108 }
109
110 std::memcpy(messageBuffer, &header, sizeof(message_header));
111 std::memcpy(messageBuffer + sizeof(message_header), &message, header.m_payloadSize);
112
113 SBK_STATUS_TRY_C(m_ringBuffer.commit_write(reserveIndex, sizeof(message_header) + header.m_payloadSize + paddingSize));
114 return SBK_SUCCESS;
115 }

117 [[nodiscard]] auto read_begin(message_view* outMessageView) noexcept -> sbk_status
118 {
119 SBK_STATUS_CHECK(outMessageView != nullptr, SBK_ERR_INVALID_PARAMETER);
120
121 std::uint8_t* messageBuffer{};
122 std::size_t readIndex{};
123 std::size_t bytesRead{};
124 numeric_message_type messageType = s_skipFlag;
125 message_size_type payloadSize{};
126
127 do
128 {
129 // READ HEADER
130
131 SBK_STATUS_TRY_C(m_ringBuffer.read_begin(&messageBuffer, &readIndex, sizeof(message_header), &bytesRead));
132 message_header* header = reinterpret_cast<message_header*>(messageBuffer);
133 messageType = header->m_type;
134 payloadSize = header->m_payloadSize;
135 SBK_STATUS_TRY_C(m_ringBuffer.read_end(sizeof(message_header)));
136
137 // AUTO READ SKIP PAYLOAD
138
139 if (header->m_type == s_skipFlag)
140 {
141 SBK_STATUS_TRY_C(m_ringBuffer.read_begin(&messageBuffer, &readIndex, payloadSize, &bytesRead));
142 SBK_STATUS_TRY_C(m_ringBuffer.read_end(payloadSize));
143 }
144
145 } while (messageType == s_skipFlag);
146
147 outMessageView->m_type = static_cast<T>(messageType);
148 outMessageView->m_payloadSize = payloadSize;
149 outMessageView->payload = messageBuffer + sizeof(message_header);
150 return SBK_SUCCESS;
151 }
