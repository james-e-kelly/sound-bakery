140 {
141 ZoneScoped;
142 SBK_STATUS_CHECK_MSG(to_numeric(type) != s_skipFlag, SBK_ERR_INVALID_PARAMETER, "Users should not create messages with a flag equal to s_skipFlag. This should only be used by the message_queue to automically wrap around the end of the buffer");
143 BOOST_ASSERT_MSG(std::numeric_limits<payload_size_t>::max() >= sizeof(U), "Message is too large for the queue. Consider increasing the payload_size_t size or decreasing the message size");
144
145 std::uint8_t* messageBuffer{};
146 std::uint8_t* paddingBuffer{};
147 std::size_t paddingSize{};
148 std::size_t reserveIndex{};
149
150 message_header header{.m_identifier = to_numeric(type), .m_payloadSize = static_cast<payload_size_t>(sizeof(U))};
151 SBK_STATUS_TRY_C(m_ringBuffer.reserve_write(&messageBuffer, sizeof(message_header) + header.m_payloadSize, &reserveIndex, &paddingBuffer, &paddingSize));
152
153 // If we were given padding, it means the buffer is split
154 // Add a skip message into end of the buffer. Write the actual message as normal
155 if (paddingSize > 0U)
156 {
157 BOOST_ASSERT(paddingSize >= sizeof(message_header));
158 message_header skipHeader{.m_identifier = s_skipFlag, .m_payloadSize = static_cast<payload_size_t>(paddingSize) - sizeof(message_header)};
159 std::memcpy(paddingBuffer, &skipHeader, sizeof(message_header));
160 }
161
162 std::memcpy(messageBuffer, &header, sizeof(message_header));
163 std::memcpy(messageBuffer + sizeof(message_header), &message, header.m_payloadSize);
164
165 SBK_STATUS_TRY_C(m_ringBuffer.commit_write(reserveIndex, sizeof(message_header) + header.m_payloadSize + paddingSize));
166 return SBK_SUCCESS;
167 }

176 [[nodiscard]] auto read_begin(message_view* outMessageView) noexcept -> sbk_status
177 {
178 ZoneScoped;
179 SBK_STATUS_CHECK(outMessageView != nullptr, SBK_ERR_INVALID_PARAMETER);
180
181 std::uint8_t* messageBuffer{};
182 std::size_t readIndex{};
183 std::size_t bytesRead{};
184 numeric_message_identifier_t messageType = s_skipFlag;
185 payload_size_t payloadSize{};
186
187 do
188 {
189 // READ HEADER
190
191 SBK_STATUS_TRY_C(m_ringBuffer.read_begin(&messageBuffer, &readIndex, sizeof(message_header), &bytesRead));
192 message_header* header = reinterpret_cast<message_header*>(messageBuffer);
193 messageType = header->m_identifier;
194 payloadSize = header->m_payloadSize;
195 SBK_STATUS_TRY_C(m_ringBuffer.read_end(sizeof(message_header)));
196
197 // AUTO READ SKIP PAYLOAD
198
199 if (header->m_identifier == s_skipFlag)
200 {
201 SBK_STATUS_TRY_C(m_ringBuffer.read_begin(&messageBuffer, &readIndex, payloadSize, &bytesRead));
202 SBK_STATUS_TRY_C(m_ringBuffer.read_end(payloadSize));
203 }
204
205 } while (messageType == s_skipFlag);
206
207 outMessageView->m_identifier = static_cast<T>(messageType);
208 outMessageView->m_payloadSize = payloadSize;
209 outMessageView->payload = messageBuffer + sizeof(message_header);
210 return SBK_SUCCESS;
211 }
