31 // ---------------------------------------------------------------------------
32 // Tracy fibers
33 //
34 // A scheduled coroutine can migrate between executor threads across its
35 // co_await hops, so ordinary per-thread zones can never show its true
36 // timeline. Tracy models this with "fibers": each coroutine gets its own lane
37 // and an executor marks when it enters (resumes) and leaves (suspends) that
38 // lane -- see @r executor::schedule_awaiter.
39 //
40 // Tracy identifies a fiber by the *address* of its name and reads the text on
41 // demand, so the name must have a stable, unique address for as long as the
42 // coroutine lives. The coroutine frame is exactly that owner, so we hang the
43 // name off the promise. Mixed into every task/detached promise; empty (and
44 // free) when Tracy is compiled out.
45 // ---------------------------------------------------------------------------
46#ifdef TRACY_ENABLE
47 struct fiber_promise
48 {
49 // Minted lazily the first time an executor schedules this coroutine, so a
50 // task that never hops threads pays nothing. The counter only has to make
51 // the lane unique; it is not tied to the source function.
52 [[nodiscard]] auto fiber_name() -> const char*
53 {
54 if (m_fiberName.empty())
55 {
56 static std::atomic<std::uint64_t> counter{0};
57 m_fiberName = fmt::format("task {}", counter.fetch_add(1, std::memory_order_relaxed));
58 }
59 return m_fiberName.c_str();
60 }
61
62 private:
63 std::string m_fiberName;
64 };
65#else
66 struct fiber_promise
67 {
68 };
69#endif

72 // via symmetric transfer. If nobody is waiting, park on noop_coroutine so

98 // Two ways to finish:
99 // co_return result<T>{...} / co_return T{...} -- normal completion
100 // co_return some_error; -- error propagation
101 // The second overload is what SBK_CO_TRY targets: on a failed await it
102 // does `co_return tmp.error();`, so an inner failure flows outward with
103 // no exception in sight.
104 auto return_value(result<T> r) noexcept -> void { outcome = std::move(r); }
105 auto return_value(error e) noexcept -> void { outcome = result<T>{e}; }
106
107 // No exceptions cross this boundary, by construction. Under
108 // -fno-exceptions the compiler never even calls this; we still declare
109 // it (the coroutine machinery requires the name) and make it fatal.
110 auto unhandled_exception() noexcept -> void { std::abort(); }
111 };

138 // Awaiting a task starts it (symmetric transfer into its body) and, when
139 // it finishes, hands back its result<T>. This is where one coroutine
140 // plugs into another.
141 struct awaiter
142 {
143 handle_type handle;
144
145 [[nodiscard]] auto await_ready() const noexcept -> bool { return false; }
146
147 [[nodiscard]] auto await_suspend(std::coroutine_handle<> caller) const noexcept -> std::coroutine_handle<>
148 {
149 handle.promise().continuation = caller;
150 return handle; // tail-call into the task body
151 }
152
153 [[nodiscard]] auto await_resume() const -> result<T>
154 {

156 }

269// Coroutine control-flow macros -- the co_return form of SBK_TRY / SBK_TRYV.
270//
271// The macros in error/result.h use `return`, which is illegal inside a coroutine
272// body. Inside a task<T> / async_result<T>, use these instead: they co_await the
273// fallible call, bind its value (or forward the error via co_return). A plain
274// sbk::result<U> also works as the operand -- operator co_await above makes it an
275// immediately-ready awaitable, so the same macro covers async and sync callees.
276// ===========================================================================

285#define SBK_CO_TRY(decl, expr) \
286 auto&& SBK_DETAIL_UNIQUE(sbkCoResult_) = co_await (expr); \
287 if (!SBK_DETAIL_UNIQUE(sbkCoResult_).has_value()) [[unlikely]] \
288 co_return ::tl::make_unexpected(std::move(SBK_DETAIL_UNIQUE(sbkCoResult_)).error()); \
289 decl = std::move(SBK_DETAIL_UNIQUE(sbkCoResult_)).value()
290
294#define SBK_CO_TRYV(expr) \
