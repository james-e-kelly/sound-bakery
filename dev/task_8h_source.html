38 // Two ways to finish:
39 // co_return result<T>{...} / co_return T{...} -- normal completion
40 // co_return some_error; -- error propagation
41 // The second overload is what SBK_CO_TRY targets: on a failed await it
42 // does `co_return tmp.error();`, so an inner failure flows outward with
43 // no exception in sight.
44 auto return_value(result<T> r) noexcept -> void { outcome = std::move(r); }
45 auto return_value(error e) noexcept -> void { outcome = result<T>{e}; }
46
47 // No exceptions cross this boundary, by construction. Under
48 // -fno-exceptions the compiler never even calls this; we still declare
49 // it (the coroutine machinery requires the name) and make it fatal.
50 auto unhandled_exception() noexcept -> void { std::abort(); }
51 };

99 // Awaiting a task starts it (symmetric transfer into its body) and, when
100 // it finishes, hands back its result<T>. This is where one coroutine
101 // plugs into another.
102 struct awaiter

105

107
108 [[nodiscard]] auto await_suspend(std::coroutine_handle<> caller) const noexcept -> std::coroutine_handle<>
109 {
110 handle.promise().continuation = caller;
111 return handle; // tail-call into the task body
112 }

115 {

203// Coroutine control-flow macros -- the co_return form of SBK_TRY / SBK_TRYV.
204//
205// The macros in error/result.h use `return`, which is illegal inside a coroutine
206// body. Inside a task<T> / async_result<T>, use these instead: they co_await the
207// fallible call, bind its value (or forward the error via co_return). A plain
208// sbk::result<U> also works as the operand -- operator co_await above makes it an
209// immediately-ready awaitable, so the same macro covers async and sync callees.
210// ===========================================================================
211
219#define SBK_CO_TRY(decl, expr) \
220 auto&& SBK_DETAIL_UNIQUE(sbkCoResult_) = co_await (expr); \
221 if (!SBK_DETAIL_UNIQUE(sbkCoResult_).has_value()) [[unlikely]] \
222 co_return ::tl::make_unexpected(std::move(SBK_DETAIL_UNIQUE(sbkCoResult_)).error()); \
223 decl = std::move(SBK_DETAIL_UNIQUE(sbkCoResult_)).value()
224
228#define SBK_CO_TRYV(expr) \
229 do \
230 { \
231 auto&& SBK_DETAIL_UNIQUE(sbkCoResult_) = co_await (expr); \
232 if (!SBK_DETAIL_UNIQUE(sbkCoResult_).has_value()) [[unlikely]] \
233 co_return ::tl::make_unexpected(std::move(SBK_DETAIL_UNIQUE(sbkCoResult_)).error()); \
234 } while (0)
