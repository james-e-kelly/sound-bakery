99 struct SB_CLASS action_update
100 {
101 template <class event_class, class state_machine_class, class source_state_class, class target_state_class>
102 auto operator()(event_class const& update, state_machine_class& stateMachine, source_state_class&, target_state_class&) -> void
103 {
104 ZoneScoped;
105
106 if (stateMachine.m_soundInstance)
107 {
108 if (ma_sound_at_end(&stateMachine.m_soundInstance->sound) == MA_TRUE &&
109 node_instance_fsm::node_group_is_idle(stateMachine.m_nodeGroup))
110 {
111 stateMachine.process_event(event_stop());
112 }
113 }
114 else if (!stateMachine.m_children.empty())
115 {
116 unsigned int stoppedChildren = 0;
117
118 for (const auto& child : stateMachine.m_children)
119 {
120 (void)child->update();
121
122 if (!child->is_playing())
123 {
124 ++stoppedChildren;
125 }
126 }
127
128 const bool allChildrenHaveStopped = stoppedChildren == stateMachine.m_children.size();
129
130 // Keep the container playing while its own DSP tail (e.g. a delay) is still ringing out
131 if (allChildrenHaveStopped && node_instance_fsm::node_group_is_idle(stateMachine.m_nodeGroup))
132 {
133 // Sequence nodes retrigger when the current sound stops
134 if (stateMachine.m_referencingNode->get_object_type() == rttr::type::get<sbk::engine::sequence_container>())
135 {
136 stateMachine.m_children.clear();
137 ++stateMachine.m_numTimesPlayed;
138 (void)stateMachine.init_child();
139
140 stateMachine.process_event(event_play());
141 }
142 else
143 {
144 stateMachine.process_event(event_stop());
145 }
146 }
147 }
148 else
149 {
150 stateMachine.process_event(event_stop());
151 }
152 }
153 };
