182 static auto align_up(std::uint8_t* p, std::size_t alignment) noexcept -> std::uint8_t*
183 {
184 if (alignment == 0U || (alignment & (alignment - 1U)) != 0U)

186 return nullptr; // non-power-of-two alignment is invalid
