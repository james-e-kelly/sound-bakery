129 template <typename T, typename... Args>
130 auto object_owner::create_owned(sbk::memory::memory_resource& resource, Args&&... args) -> sbk::result<sbk::owned_ptr<T>>
131 {
132 static_assert(std::is_base_of_v<object_owner, T>, "create_owned is for graph nodes; they must derive from object_owner");
133
134 ZoneScoped;
135
136 void* const memory = resource.allocate(sizeof(T), alignof(T));
137 SBK_CHECK(memory != nullptr, SBK_ERR_OUT_OF_MEMORY);
138
139 try
140 {
141 T* const raw = ::new (memory) T(std::forward<Args>(args)...);
142 raw->set_owner(this); // owned objects still know their parent/owner
143 return sbk::owned_ptr<T>(raw, sbk::memory::owned_object_deleter<T>(resource));
144 }
145 catch (const std::exception& exception)
146 {
147 // Constructor threw: the placement-new never completed, so return the raw
148 // block ourselves -- no destructor to run. Mirrors create_runtime_object.
149 resource.deallocate(memory, sizeof(T), alignof(T));
150 return sbk::make_error(SBK_ERR_OUT_OF_MEMORY, exception.what());
151 }
152 }
