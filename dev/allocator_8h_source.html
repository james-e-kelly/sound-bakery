155
156 auto deallocate(void* pointer, size_t size) noexcept -> void
157 {
158 if (pointer == nullptr)
159 {
160 return;
161 }
162 if (memory_resource* const resource = resolve())
163 {
164 resource->deallocate(pointer, size);
165 return;
166 }
167 // The resource that produced this block is gone (system torn down while a container
168 // still holds memory). Leaking here is preferable to touching freed state.
169 BOOST_ASSERT_MSG(false, "deallocation after sbk::engine::system::destroy()");
170 }

178 }

180 }
