59 [[nodiscard]] auto init(std::size_t size, sbk::memory::memory_resource& allocator) noexcept -> sbk::result<>
60 {
61 SBK_CHECK(m_buffer == nullptr, SBK_ERR_ALREADY_INITIALIZED);
62 SBK_CHECK(m_capacity == 0, SBK_ERR_ALREADY_INITIALIZED);
63 SBK_CHECK_MSG(size > 2, SBK_ERR_INVALID_PARAMETER, "Size was too small");
64 SBK_CHECK_MSG(decltype(m_readIndex)::is_always_lock_free, SBK_ERR_SYSTEM, "Atomic was found to not be lock free");
65
66 m_capacity = std::bit_ceil(size);
67 m_mask = m_capacity - 1;
68 SBK_CHECK(sbk::memory::is_pow_2(m_capacity), SBK_ERR_BAKERY);
69
70 m_buffer = static_cast<std::uint8_t*>(allocator.allocate(m_capacity, sbk::memory::default_alignment));
71 SBK_CHECK(m_buffer != nullptr, SBK_ERR_OUT_OF_MEMORY);
72
73 m_memoryResource = &allocator;
74
75 m_reserveWriteIndex.store(0, std::memory_order_relaxed);
76 m_committedWriteIndex.store(0, std::memory_order_relaxed);
77 m_readIndex.store(0, std::memory_order_relaxed);
78
79 return sbk::ok();
80 }

93 [[nodiscard]] auto reserve_write(std::uint8_t** outBuffer, std::size_t size, std::size_t* outReserveIndex, std::uint8_t** outPadding, std::size_t* outPaddingSize) noexcept -> sbk_status
94 {
95 SBK_STATUS_CHECK(outBuffer != nullptr && outReserveIndex != nullptr && outPadding != nullptr && outPaddingSize != nullptr && size != 0, SBK_ERR_INVALID_PARAMETER);
96 SBK_STATUS_CHECK(m_buffer != nullptr && m_capacity > 0, SBK_ERR_UNINITIALIZED);
97 SBK_STATUS_CHECK(size <= m_capacity, SBK_ERR_INVALID_PARAMETER);
98
99 *outPadding = nullptr;
100 *outPaddingSize = 0U;
101
102 for (;;)
103 {
104 // Acquire pairs with the reader's release-store on m_readIndex.
105 const std::size_t read = m_readIndex.load(std::memory_order_acquire);
106 std::size_t reserveWrite = m_reserveWriteIndex.load(std::memory_order_relaxed);
107
108 const std::size_t writeOffset = reserveWrite & m_mask;
109 const std::size_t spaceFromWriteTillEnd = m_capacity - writeOffset;
110 const bool needsWrapping = size > spaceFromWriteTillEnd;
111
112 // Wrap reservations pay for the end-of-buffer padding too.
113 const std::size_t reserveSize = needsWrapping ? (size + spaceFromWriteTillEnd) : size;
114
115 SBK_STATUS_CHECK(can_reserve_bytes(reserveWrite, read, reserveSize), SBK_ERR_FULL);
116
117 if (m_reserveWriteIndex.compare_exchange_weak(reserveWrite, reserveWrite + reserveSize, std::memory_order_relaxed))
118 {
119 *outReserveIndex = reserveWrite;
120
121 if (needsWrapping)
122 {
123 *outBuffer = m_buffer;
124 *outPadding = m_buffer + writeOffset;
125 *outPaddingSize = spaceFromWriteTillEnd;
126 }
127 else
128 {
129 *outBuffer = m_buffer + writeOffset;
130 }
131 return SBK_SUCCESS;
132 }
133 }
134 }

145 [[nodiscard]] auto commit_write(std::size_t reserveIndex, std::size_t size) noexcept -> sbk_status
146 {
147 SBK_STATUS_CHECK(m_buffer != nullptr && m_capacity > 0, SBK_ERR_UNINITIALIZED);
148
149 // Publish in reserve order - see the commit-ordering note on the class.
150 std::size_t expected = reserveIndex;
151 while (!m_committedWriteIndex.compare_exchange_weak(expected, reserveIndex + size, std::memory_order_release, std::memory_order_relaxed))
152 {
153 expected = reserveIndex;
154 std::this_thread::yield();
155 }
156 return SBK_SUCCESS;
157 }

168 [[nodiscard]] auto write(const void* message, std::size_t size) noexcept -> sbk_status
169 {
170 SBK_STATUS_CHECK(message != nullptr && size != 0, SBK_ERR_INVALID_PARAMETER);
171 SBK_STATUS_CHECK(m_buffer != nullptr && m_capacity > 0, SBK_ERR_UNINITIALIZED);
172 SBK_STATUS_CHECK(size <= m_capacity, SBK_ERR_INVALID_PARAMETER);
173
174 for (;;)
175 {
176 const std::size_t read = m_readIndex.load(std::memory_order_acquire);
177 std::size_t reserveWrite = m_reserveWriteIndex.load(std::memory_order_relaxed);
178
179 SBK_STATUS_CHECK(can_reserve_bytes(reserveWrite, read, size), SBK_ERR_FULL);
180
181 if (m_reserveWriteIndex.compare_exchange_weak(reserveWrite, reserveWrite + size, std::memory_order_relaxed))
182 {
183 const std::size_t writeOffset = reserveWrite & m_mask;
184 const std::size_t firstChunkSize = std::min(size, m_capacity - writeOffset);
185 const bool needsWrapping = firstChunkSize < size;
186
187 std::memcpy(m_buffer + writeOffset, message, firstChunkSize);
188 if (needsWrapping)
189 {
190 std::memcpy(m_buffer, static_cast<const std::uint8_t*>(message) + firstChunkSize, size - firstChunkSize);
191 }
192
193 std::size_t expected = reserveWrite;
194 while (!m_committedWriteIndex.compare_exchange_weak(expected, reserveWrite + size, std::memory_order_release, std::memory_order_relaxed))
195 {
196 expected = reserveWrite;
197 std::this_thread::yield();
198 }
199 return SBK_SUCCESS;
200 }
201 }

214 {
215 SBK_STATUS_CHECK(outBuffer != nullptr && outReadIndex != nullptr && outActualSize != nullptr && size != 0, SBK_ERR_INVALID_PARAMETER);

252 {
253 SBK_STATUS_CHECK(outBuffer != nullptr && size != 0, SBK_ERR_INVALID_PARAMETER);
254 SBK_STATUS_CHECK(m_buffer != nullptr && m_capacity > 0, SBK_ERR_UNINITIALIZED);
255 SBK_STATUS_CHECK(size <= m_capacity, SBK_ERR_INVALID_PARAMETER);
256

258 const std::size_t committedWrite = m_committedWriteIndex.load(std::memory_order_acquire);
259
260 SBK_STATUS_CHECK(!is_buffer_empty(committedWrite, read), SBK_ERR_EMPTY);
261 SBK_STATUS_CHECK(can_read_bytes(committedWrite, read, size), SBK_ERR_TOO_LARGE);
262

264 const std::size_t firstChunkSize = std::min(size, m_capacity - readOffset);
